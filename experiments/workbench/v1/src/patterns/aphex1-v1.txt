// aphex1 — original fast broken-rhythm sketch, 2026-09-28
// Preserved unchanged so the first listening draft remains selectable.
// TEMPO is cycles per minute: 27 cycles = about 108 beats per minute in 4/4.
setcpm(27)
const PAD = 0.38
const SUB = 0.58
const DRUMS = 0.86
const GHOST = 0.18
stack(
  s("pad").slow(4).gain(PAD).lpf("<600 1400 900 2200>").room(0.25),
  s("sub").slice(8, "0 ~ 2 ~ 4 ~ 6 ~").gain(SUB).lpf(240),
  s("drums").slice(16, "<0 0 8 0> 2 [3 3] 5 4 7 8 [10 11] 12 14 15 ~").gain(DRUMS),
  s("drums").slice(16, "~ ~ ~ [11 11] ~ ~ 15 ~").speed(1.4).gain(GHOST)
)

const smooth = x => { x=Math.max(0,Math.min(1,x)); return x*x*(3-2*x) }
const frame = time => {
 const t=Math.max(0,Number(time)), c=Math.min(15,Math.floor(t/16)), local=t%16
 const now=CHAPTERS[c], before=CHAPTERS[Math.max(0,c-1)]
 const f={chapter:c,local,name:now[0]}, names=['kick','snare','hh','oh','acid','response','perc','ride','pad','cutoff','q','drive']
 for(let k=0;k<names.length;k++) f[names[k]]=before[k+1]+(now[k+1]-before[k+1])*smooth(local/2)
 // Kick changes are exact at chapter boundaries; avoid a half-hearted downbeat.
 f.kick=now[1]
 // Eight-bar questions and answers; peaks breathe rather than getting louder every beat.
 const breath=.83+.17*Math.sin((t-2)*Math.PI/8)
 f.cutoff*=breath*(.88+.24*local/16)
 f.decay=.12+.055*smooth(local/16)+.025*(c===7||c===12)
 // A short drum vacuum before both breakdowns and the final return.
 if((c===3||c===7)&&local>=15) {f.kick=0;f.snare=0;f.oh=0;f.ride=0}
 // The second break grows, then leaves half a bar before the return.
 if(c===9) {f.cutoff=640+1800*smooth(local/16);f.q=19+3*smooth(local/16);if(local>=15.5){f.hh=0;f.acid=0;f.response=0}}
 // First four bars introduce the motif one fragment at a time.
 if(c===0&&local<4) f.acid*=.5
 // Last 8 bars dismantle the drums; last 4 bars hold only the pad and echo.
 if(c===15&&local>=8) {f.snare=0;f.oh=0;f.ride=0;f.perc=0;f.hh*=1-smooth((local-8)/4)}
 if(c===15&&local>=12) {f.kick=0;f.acid=0;f.response=0}
 f.end=t<256?1:0
 f.fill=t<252&&Math.floor(t)%8===7&&f.kick>0
 f.crash=[80,96,112,160,176,192].includes(Math.floor(t))
 return f
}
const motif=time=>{
 const t=Number(time), step=Math.floor((t%1)*16+1e-7)%16, chapter=Math.floor(t/16)
 const ns=[...BASE]
 // The open set's home notes remain; only weak beats answer or climb.
 if(Math.floor(t)%8>=4){
  const variants=[[3,41],[7,43],[11,38],[15,39]]
  const v=variants[(chapter+SEED)%4]; ns[v[0]]=v[1]
 }
 if(chapter===9&&Math.floor(t)%4>=2){ns[7]=48;ns[15]=46}
 if(chapter===11||chapter===12){if(Math.floor(t)%8>=6){ns[3]=43;ns[11]=46}}
 const n=ns[step], prev=ns[(step+15)%16]
 const ints=(step%2?[3,7]:[7,3]).filter(i=>[0,2,3,5,7,10].includes((n+i)%12))
 return {note:n,slide:[3,7,11].includes(step)&&n!=null&&prev!=null?n-prev:0,
  answer:n==null?null:n+12+ints[0],accent:step%4===0||step===6?1.16:.86}
}

const cutoff=signal(t=>frame(t).cutoff)
// Tags identify the six converted roles without adding onset events.
const synthPart=(pattern,part)=>pattern.withHap(h=>h.withValue(v=>({...v,afterimageSynthPart:part})))

  .withHap(h=>h.withValue(v=>({...v,note:motif(h.whole.begin).answer})))

  .attack(.002).decay(.18).sustain(0).release(.02).gain(.078).pan(.35).filterWhen(t=>frame(t).fill),'perc'),

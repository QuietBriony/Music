// Acid 303 / 909 — hard faders / ACID_FADER_BANK_V1
// Play後は画面の縦フェーダーで操作。値はこのコードにも残る。
// 909はStrudel標準バンク、303風ベースはブラウザ合成。音量に注意。
const CUTOFF = slider(420, 120, 2200, 10) // フィルターの開き
const RESONANCE = slider(18, 1, 35, 1) // みょんの強さ
const DRIVE = slider(3.2, 0, 7, 0.1) // 歪み
const DECAY = slider(0.18, 0.03, 0.5, 0.01) // うねりの長さ
setcpm(32)
stack(
  s("bd*4").bank("RolandTR909").gain(0.55),
  s("~ sd ~ sd").bank("RolandTR909").gain(0.34),
  s("~ hh ~ hh ~ hh ~ hh").bank("RolandTR909").gain(0.22),
  s("~ ~ ~ ~ ~ ~ ~ oh").bank("RolandTR909").gain(0.12),
  note("<[c2 ~ c2 eb2 g1 ~ bb1 c2] [c2 ~ eb2 g1 c2 ~ d2 bb1] [c2 c3 ~ eb2 g1 ~ f2 eb2] [c2 ~ c2 eb2 ~ g1 bb1 d2]>")
    .s("sawtooth")
    .attack(0.003).decay(0.13).sustain(0).release(0.03)
    .lpf(CUTOFF).lpq(RESONANCE)
    .lpa(0.003).lpd(DECAY).lps(0).lpenv(6.5)
    .ftype("ladder")
    .distort(DRIVE).postgain(0.48)
    .gain(0.36)
)

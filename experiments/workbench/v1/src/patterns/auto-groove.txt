// Auto Groove — bounded algorithmic variation, 2026-09-29
// Kick stays predictable; quiet high percussion randomly drops out each cycle.
// Every eighth cycle leaves a short kick break. This is pattern generation, not AI listening.
// 29 cycles per minute = about 116 BPM in 4/4.
setcpm(29)
const KICK = 0.7
const HATS = 0.25
const GHOST = 0.12
const SUB = 0.5
const PAD = 0.24
stack(
  s("drums").slice(16, "0 0 0 0").clip(0.78).lpf(250).gain(KICK)
    .mask("<1 1 1 1 1 1 1 0>"),
  s("drums").slice(16, "~ 1 ~ 3 ~ 5 ~ 15").clip(0.3).hpf(1700).gain(HATS)
    .degradeBy(0.3),
  s("drums").slice(16, "~ ~ ~ 11 ~ ~ 15 ~").clip(0.28).hpf(1000).gain(GHOST)
    .degradeBy(0.7),
  s("sub").slice(8, "<0 0 2 0> ~ 4 ~ 0 ~ 6 ~").clip(0.8).lpf(220).gain(SUB),
  s("pad").slow(4).lpf("<500 900 1450 700 1200 550 1800 850>").room(0.24).gain(PAD)
)

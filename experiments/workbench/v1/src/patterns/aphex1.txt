// aphex1 — sparse-drum revision, 2026-09-28
// Listening feedback: give the chopped drums more space and remove the fast ghost layer.
// 24 cycles per minute = about 96 beats per minute in 4/4.
setcpm(24)
const PAD = 0.38
const SUB = 0.58
const DRUMS = 0.68
stack(
  s("pad").slow(4).gain(PAD).lpf("<600 1400 900 2200>").room(0.25),
  s("sub").slice(8, "0 ~ 2 ~ 4 ~ 6 ~").gain(SUB).lpf(240),
  s("drums").slice(16, "0 ~ ~ ~ 4 ~ 6 ~ 8 ~ ~ ~ 12 ~ 14 ~").gain(DRUMS)
)
